import DocumentUpload from '../components/DocumentUpload'
import { useState } from 'react'
import { supabase } from '../lib/supabase'
import { toast } from 'react-hot-toast'
import { Upload, FileText, X, AlertCircle } from 'lucide-react'

interface Props {
  criancaId: string
  onUploadComplete: (url: string) => void
}

export default function DocumentUpload({ criancaId, onUploadComplete }: Props) {
  const [uploading, setUploading] = useState(false)
  const [progress, setProgress] = useState(0)

  const handleFile = async (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0]
    if (!file) return

    // Validações
    const allowedTypes = ['application/pdf', 'image/jpeg', 'image/png']
    if (!allowedTypes.includes(file.type)) {
      toast.error('Apenas PDF, JPG ou PNG são permitidos')
      return
    }
    if (file.size > 5 * 1024 * 1024) {
      toast.error('Ficheiro máximo: 5MB')
      return
    }

    setUploading(true)
    setProgress(0)

    try {
      // Nome único seguro
      const ext = file.name.split('.').pop()
      const fileName = `${criancaId}/${Date.now()}-${Math.random().toString(36).slice(2, 8)}.${ext}`

      // Simulação de progresso (Supabase JS não expõe progresso nativo de upload)
      const progressInterval = setInterval(() => {
        setProgress(prev => (prev < 90 ? prev + 10 : prev))
      }, 200)

      const { data, error } = await supabase.storage
        .from('origem_creative_documentos')
        .upload(fileName, file, { cacheControl: '3600', upsert: false })

      clearInterval(progressInterval)
      setProgress(100)

      if (error) throw error

      // Gera URL pública temporária (bucket privado)
      const { data: urlData } = await supabase.storage
        .from('origem_creative_documentos')
        .createSignedUrl(data.path, 60 * 60 * 24 * 30) // 30 dias

      onUploadComplete(urlData?.signedUrl || '')
      toast.success('Documento carregado com sucesso')
    } catch (err: any) {
      toast.error(err.message || 'Erro ao carregar documento')
    } finally {
      setUploading(false)
      setProgress(0)
      e.target.value = '' // limpa input
    }
  }

  return (
    <div className="border-2 border-dashed border-gray-300 rounded-lg p-6 text-center bg-gray-50">
      <input
        type="file"
        accept=".pdf,.jpg,.jpeg,.png"
        onChange={handleFile}
        disabled={uploading}
        className="hidden"
        id="doc-upload"
      />
      <label htmlFor="doc-upload" className="cursor-pointer flex flex-col items-center gap-2">
        {uploading ? (
          <>
            <Upload className="animate-spin text-blue-600" size={32} />
            <span className="text-sm text-gray-600">A carregar... {progress}%</span>
            <div className="w-full bg-gray-200 rounded-full h-2 mt-2">
              <div className="bg-blue-600 h-2 rounded-full transition-all" style={{ width: `${progress}%` }}></div>
            </div>
          </>
        ) : (
          <>
            <FileText size={32} className="text-gray-400" />
            <span className="text-sm font-medium text-gray-700">Clica para carregar documento</span>
            <span className="text-xs text-gray-400">PDF, JPG ou PNG (máx. 5MB)</span>
          </>
        )}
      </label>
    </div>
  )
}
{criancaId && (
  <div className="col-span-2 mt-2">
    <label className="block text-sm font-medium text-gray-700 mb-2">Documentos Anexados</label>
    <DocumentUpload 
      criancaId={criancaId} 
      onUploadComplete={(url) => console.log('📎 URL seguro:', url)} 
    />
  </div>
)}
